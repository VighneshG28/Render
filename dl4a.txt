import numpy as np
import tensorflow as tf

from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.utils import to_categorical
from tensorflow.keras import Sequential, Input
from tensorflow.keras.layers import Embedding, Lambda, Dense

from sklearn.metrics.pairwise import euclidean_distances

data = [
    "I like natural language processing.",
    "Computers understand human language.",
    "Word embeddings represent words.",
    "CBOW predicts a target word.",
    "Skip Gram is another model.",
    "Word2Vec is used in NLP.",
    "Deep learning uses embeddings.",
    "CBOW works well with common words."
]

tokenizer = Tokenizer()
tokenizer.fit_on_texts(data)

word2id = tokenizer.word_index
id2word = {v: k for k, v in word2id.items()}

corpus = tokenizer.texts_to_sequences(data)

vocab_size = len(word2id) + 1
window_size = 2
embed_size = 100

def generate_pairs(corpus):
    for sentence in corpus:
        for i, target in enumerate(sentence):

            context = [
                sentence[j]
                for j in range(
                    max(0, i - window_size),
                    min(len(sentence), i + window_size + 1)
                )
                if j != i
            ]

            x = pad_sequences(
                [context],
                maxlen=window_size * 2
            )

            y = to_categorical([target], vocab_size)

            yield x, y

cbow = Sequential([
    Input(shape=(window_size * 2,)),
    Embedding(vocab_size, embed_size),
    Lambda(lambda x: tf.reduce_mean(x, axis=1)),
    Dense(vocab_size, activation="softmax")
])

cbow.compile(
    optimizer="adam",
    loss="categorical_crossentropy"
)

for epoch in range(10):
    total_loss = 0

    for x, y in generate_pairs(corpus):
        total_loss += float(cbow.train_on_batch(x, y))

    print(f"Epoch {epoch + 1}: Loss = {total_loss:.3f}")

weights = cbow.layers[0].get_weights()[0][1:]

distance_matrix = euclidean_distances(weights)

for word in ["deep", "cbow"]:

    index = word2id[word] - 1

    nearest = distance_matrix[index].argsort()[1:6]

    similar = [id2word[i + 1] for i in nearest]

    print(word, ":", similar)
